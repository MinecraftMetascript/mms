Build-time loop, if and let

Const biomes = ["meadow", "forest"]

for i, b in biomes {
  let r = Rand("shrine/${b}")
  Function "shrine/enter_${b}" {
    /say ${b}
  }
  if (b != "forest") {
    BiomeTag "has/${b}" = [b]
  } else {
    OnLoad {
      /say forest
    }
  }
}

Assert "biomes are unique" = Unique(biomes)
